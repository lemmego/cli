// Bound by c.ParseInput, which picks a decoder from the request's
// Content-Type: `json:` for a JSON body, `form:` for a form — URL-encoded,
// multipart, or a query string on a request with no body.
//
// A file is a *multipart.FileHeader, the standard library's own type: it
// carries the filename, the size and the headers, and Open() gives the
// content. Use []*multipart.FileHeader for a field that takes several, which
// binds a repeated key, `name[0]` and `name[]` alike.
type {{.InputName | toCamel}}Input struct {
    ctx      app.Context
{{- range .Fields}}
    {{- if eq .Type "file"}}
    {{.Name | toCamel}} *multipart.FileHeader `json:"-" form:"{{.Name | toSnake}}"`
    {{- else}}
    {{.Name | toCamel}} {{.Type}} `json:"{{.Name | toSnake}}" form:"{{.Name | toSnake}}"`
    {{- end}}
{{- end}}
}

func New{{.InputName | toCamel}}Input(c app.Context) (*{{.InputName | toCamel}}Input, error) {
    input := &{{.InputName | toCamel}}Input{}
    err := c.ParseInput(input)
    if err != nil {
        return nil, err
    }
    input.ctx = c
    return input, input.Validate()
}

func (i *{{.InputName | toCamel}}Input) Validate() error {
    v := i.ctx.Validator()
    {{- range .Fields}}

    v.Field("{{.Name | toSnake}}", i.{{.Name | toCamel}}).Required().
        Check(db.Unique(i.ctx.App(), "{{.Table}}", "{{.Name | toSnake}}"))
        {{- else if .Required}}
    v.Field("{{.Name | toSnake}}", i.{{.Name | toCamel}}).Required()
        {{- else if .Unique}}
    v.Field("{{.Name | toSnake}}", i.{{.Name | toCamel}}).
        Check(db.Unique(i.ctx.App(), "{{.Table}}", "{{.Name | toSnake}}"))
        {{- end}}
    {{- end}}
    // Err is a rule that could not run — a failed uniqueness lookup, say —
    // as opposed to a value the person can fix. Reporting it separately lets
    // the caller answer 503 rather than telling someone their value is taken
    // when the database is simply unreachable.
    if err := v.Err(); err != nil {
        return err
    }
    return v.Validate()
}
